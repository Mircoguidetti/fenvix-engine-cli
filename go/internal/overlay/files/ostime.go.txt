// Code added by fenvix-go (build-time overlay, --det). Not part of Go.
//
// Real time seen next to the virtual clock (review R-025 D2):
//   * file modification times from Stat, Lstat and File.Stat are moved by
//     the offset between the virtual and the real clock at the start of the
//     test, so a file written during the test is as old on the virtual clock
//     as it is in real time since the test started (cache TTLs, staleness
//     checks), and older files stay in the virtual past;
//   * programs the test starts get the same offset for their wall clock
//     through a preloaded library (dynamically linked programs only).
//   * Chtimes moves the times it sets the other way, so a time set and read
//     back is unchanged.
// Not moved: access and change times through Sys(), File.Stat's raw times
// set by syscall.UtimesNano.

package os

import (
	"runtime"
	"syscall"
	"time"
)

func fenvixModTime(t time.Time) time.Time {
	if off, ok := syscall.Fenvix_RealOffset(); ok && runtime.Fenvix_UserG() {
		return t.Add(time.Duration(off))
	}
	return t
}

// fenvixChtimes moves a time the program sets back to the real timeline, so
// that Chtimes then Stat returns it unchanged (rclone bisync, found in the
// flaky round flaky-r8). The zero time (leave unchanged) is kept.
func fenvixChtimes(t time.Time) time.Time {
	if t.IsZero() {
		return t
	}
	if off, ok := syscall.Fenvix_RealOffset(); ok && runtime.Fenvix_UserG() {
		return t.Add(-time.Duration(off))
	}
	return t
}

func fenvixChildAttr(attr *ProcAttr) *ProcAttr {
	add := syscall.Fenvix_ChildClockEnv()
	if add == nil || !runtime.Fenvix_UserG() {
		return attr
	}
	a := ProcAttr{}
	if attr != nil {
		a = *attr
	}
	env := a.Env
	if env == nil {
		env = syscall.Environ()
	}
	out := make([]string, 0, len(env)+2)
	pre := ""
	for _, kv := range env {
		switch {
		case len(kv) >= 11 && kv[:11] == "LD_PRELOAD=":
			pre = kv[11:]
		case len(kv) >= 23 && kv[:23] == "FENVIX_GO_CLOCK_OFFSET=":
		default:
			out = append(out, kv)
		}
	}
	lp := add[1]
	if pre != "" {
		lp += ":" + pre
	}
	a.Env = append(append(out, add[0]), lp)
	return &a
}
